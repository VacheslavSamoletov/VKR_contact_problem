# -*- coding: utf-8 -*-
"""
сборка матрицы МГЭ и расчёт контактных напряжений
            на основе kernel_fgm_A.py (RK4 везде).

Матричные элементы:
    A_{ji} = J(ξ_i - x_1^j) - J(ξ_{i-1} - x_1^j),  j, i = 1, …, N

Для равномерной сетки ξ_k = -a + k·h, x_1^j = -a + (j-1/2)·h:
    ξ_i - x_1^j      = h·(i - j + 1/2)
    ξ_{i-1} - x_1^j  = h·(i - j - 1/2)
Матричный элемент зависит только от m = i - j → матрица Тёплицева;
по чётности ядра k(x) она ещё и симметричная: a_m = a_{-m}.
Достаточно вычислить J(u) в N точках u_k = h·(k + 1/2), k = 0,…,N-1.

подынтегральная функция J(u) приходит из
kernel_fgm_A.Integral
Каждое значение J(u) считается заново 



"""

import math
import time

import numpy as np
import matplotlib.pyplot as plt
from scipy.linalg import toeplitz

from kernel_fgm_A import FGMLayer, Integral


# =====================================================================
# Сборка матрицы и решение
# =====================================================================

def assemble_matrix(integral: Integral,
                    N: int,
                    a_half: float,
                    verbose: bool = False):
    """
    Собирает симметричную Тёплицеву матрицу A размера N×N.
    Возвращает (A, ys, h, J_vals), где J_vals — значения J(u_k) в узлах,
    которые могут пригодиться для отладки или повторного использования.
    """
    h = 2.0 * a_half / N
    ys = h * (np.arange(N) + 0.5)   # точки u_k = h·(k + 1/2)

    if verbose:
        print(f"Расчёт J(u) в {N} точках (h = {h:.4f}, "
              f"каждая точка = ~1000 задач Коши)...")
        t0 = time.time()

    J_vals = np.empty(N, dtype=complex)
    for k, u in enumerate(ys):
        J_vals[k] = integral.evaluate(float(u))
        if verbose and (k + 1) % 10 == 0:
            print(f"  {k+1}/{N}  ({time.time() - t0:.1f}s)")

    if verbose:
        print(f"  все J(u) посчитаны за {time.time() - t0:.1f}s")

    # Тёплицев первый столбец:
    c = np.empty(N, dtype=complex)
    c[0] = 2.0 * J_vals[0]
    c[1:] = J_vals[1:] - J_vals[:-1]

    # комплексно-симметричная Тёплицева матрица (A = A^T, не эрмитова).
    A = toeplitz(c, c)

    return A, ys, h, J_vals


def solve_contact(integral: Integral,
                  N: int,
                  a_half: float,
                  delta: complex = 1.0 + 0j,
                  verbose: bool = False):
    """
    Решает СЛАУ A·τ = δ·1 и возвращает результаты.
    """
    A, ys, h, J_vals = assemble_matrix(integral, N, a_half, verbose=verbose)

    rhs = delta * np.ones(N, dtype=complex)
    tau = np.linalg.solve(A, rhs)

    x_centers = -a_half + h * (np.arange(N) + 0.5)
    # 1-норма устойчивее SVD на крупных комплексных матрицах
    try:
        cond = np.linalg.cond(A, p=1)
    except np.linalg.LinAlgError:
        cond = float('nan')
    palindrome_err = np.max(np.abs(tau - tau[::-1])) \
                     / max(np.max(np.abs(tau)), 1e-30)

    if verbose:
        idx_c = N // 2
        idx_e = N - 1
        print(f"\nРешение для N = {N}:")
        print(f"  τ(в центре)   = {tau[idx_c]:+.4e}")
        print(f"  τ(у края)     = {tau[idx_e]:+.4e}")
        print(f"  |τ(центр)|    = {abs(tau[idx_c]):.4e}")
        print(f"  |τ(у края)|   = {abs(tau[idx_e]):.4e}")
        print(f"  cond(A)       = {cond:.2e}")
        print(f"  ||τ - τ_rev||/||τ|| = {palindrome_err:.2e}")

    if palindrome_err > 1e-8:
        import warnings
        warnings.warn(
            f"Решение τ не палиндромно (отн.ошибка {palindrome_err:.2e}). "
            f"Возможно нарушена симметрия матрицы.")

    return {
        'tau': tau,
        'x_centers': x_centers,
        'A': A,
        'h': h,
        'cond': cond,
        'J_vals': J_vals,
    }

def compute_stamp_displacement(tau_0, h, kappa, T, mass):
    """
    Превращает τ_0 — решение уравнения с правой частью 1 —
    в физическое τ для заданных параметров штампа T (приложенная сила)
    и m (масса штампа).

    Уравнение движения штампа в безразмерных переменных:
        -m·δ·κ² = T - ∫_{-a}^{a} τ(ξ) dξ.

    Поскольку задача линейна, τ = δ·τ_0, и подстановка даёт
        δ = T / (∫τ_0(ξ) dξ - m·κ²).

    Интеграл от кусочно-постоянного τ_0 на сетке шага h:
        ∫τ_0(ξ)dξ ≈ h · Σ τ_0[j].
    """
    integral_tau_0 = h * np.sum(tau_0)
    delta = T / (integral_tau_0 - mass * kappa ** 2)
    tau = delta * tau_0
    return delta, tau, integral_tau_0

# Демо
# =====================================================================

def main():
    print("=" * 70)
    print("BEM A: контактные напряжения с эталонным ядром (RK4 везде)")
    print("=" * 70)

    # ---------- ФГМ-слой ----------
    kappa = 7.5
    eps = 1e-3
    mu_func = lambda x: 1.0 - 0.8 * x       # «мягкий верх»
    rho_func = lambda x: 1.0

    layer = FGMLayer(mu_func=mu_func, rho_func=rho_func,
                     kappa=kappa, epsilon=eps, n_steps=200)
    print(f"ФГМ-слой:")
    print(f"  μ(0) = {mu_func(0):.2f},  μ(1) = {mu_func(1):.2f}")
    print(f"  μ_min = {layer.mu_min:.3f}")
    print(f"  κ = {kappa},  ε = {eps}")

    alpha_end = 1.2 * kappa / math.sqrt(layer.mu_min)

    #  при численном переполнении RK4 на очень больших α
    # (когда решения растут как e^α и double-precision переполняется)
    # возвращаем 0 вместо NaN. Это эквивалентно адаптивному усечению хвоста
    # там, где K_over_alpha перестаёт быть вычислимой. Ошибка усечения
    # пренебрежимо мала: |K̃| ~ 1/(μ(H)α) при α~500 порядка 10^-2.
    def safe_integrand(alpha):
        try:
            K = layer.K_over_alpha(alpha)
            if not (math.isfinite(K.real) and math.isfinite(K.imag)):
                return 0j
            return K
        except (OverflowError, ValueError):
            return 0j

    integral = Integral(safe_integrand, kappa=kappa,
                        alpha_end=alpha_end, n_sub=10)
    print(f"  α_end = {alpha_end:.3f}")

    # ---------- Параметры штампа ----------
    a_half = 0.5
    delta = 1.0 + 0j

    # ---------- Тест сходимости по N ----------
    # При желании добавьте 160 — это ещё ~60 секунд
    Ns = [20, 40, 60]
    T = 1.0 + 0j
    mass = 0.1

    print(f"\n{'=' * 82}")
    print(f"Тест сходимости физического τ для a = {a_half}, "
          f"T = {T}, m = {mass}")
    print(f"{'=' * 82}")
    print(f"{'N':>5}  {'|τ(0)|':>12}  {'|τ(0.9a)|':>12}  {'|δ|':>12}  "
          f"{'отн.изм.|τ(0)|':>16}  {'время, с':>10}")
    print("-" * 82)

    results = {}
    tau_prev = None
    total_start = time.time()

    for N in Ns:
        t0 = time.time()

        # 1. Решаем нормированную задачу — получаем τ_0
        res = solve_contact(integral, N, a_half,
                            delta=1.0 + 0j, verbose=False)

        # 2. Рассчитываем физическое смещение δ и итоговое τ
        delta_N, tau_N, integral_N = compute_stamp_displacement(
            res['tau'], res['h'], kappa, T, mass)
        res['tau_0'] = res['tau'].copy()   # сохраняем нормированное τ_0
        res['tau'] = tau_N                  # перезаписываем физическим τ
        res['delta'] = delta_N
        res['integral_tau_0'] = integral_N

        elapsed = time.time() - t0

        # 3. Метрики сходимости — теперь по физическому τ
        idx0 = np.argmin(np.abs(res['x_centers']))
        tau_center = abs(tau_N[idx0])
        idx9 = np.argmin(np.abs(res['x_centers'] - 0.9 * a_half))
        tau_edge = abs(tau_N[idx9])

        rel_change = (abs(tau_center - tau_prev) / max(tau_prev, 1e-30)
                      if tau_prev is not None else float('nan'))
        rel_str = f"{rel_change:.3e}" if not np.isnan(rel_change) else "—"

        print(f"{N:>5d}  {tau_center:>12.6e}  {tau_edge:>12.6e}  "
              f"{abs(delta_N):>12.6e}  {rel_str:>16}  {elapsed:>10.1f}")

        tau_prev = tau_center
        results[N] = res

    print(f"\nИтого: {time.time() - total_start:.1f} с")

    # Дополнительный диагностический вывод для самого мелкого N
    N_best = max(results.keys())
    res = results[N_best]
    print(f"\n{'=' * 70}")
    print(f"Диагностика для самого точного решения (N = {N_best})")
    print(f"{'=' * 70}")
    print(f"  ∫τ_0(ξ)dξ              = {res['integral_tau_0']:+.4e}")
    print(f"  m·κ²                   = {mass * kappa**2:+.4e}")
    print(f"  знаменатель формулы    = "
          f"{res['integral_tau_0'] - mass * kappa**2:+.4e}")
    print(f"  δ                      = {res['delta']:+.4e}")
    print(f"  |δ|                    = {abs(res['delta']):.4e}")
    print(f"  arg(δ)                 = {np.angle(res['delta']):+.4f} рад")
    
    
    
    
    
    
    
    
    
    # ---------- Графики ----------
    fig, axes = plt.subplots(2, 2, figsize=(14, 10))
    colors = ['#2E86AB', '#A23B72', '#08527F', '#FF8C42']

    for (N, res), color in zip(results.items(), colors):
        x = res['x_centers']
        tau = res['tau']
        axes[0, 0].plot(x, tau.real, color=color, lw=1.5,
                        marker='o', markersize=3, label=f'N = {N}')
        axes[0, 1].plot(x, tau.imag, color=color, lw=1.5,
                        marker='o', markersize=3, label=f'N = {N}')
        axes[1, 0].plot(x, np.abs(tau), color=color, lw=1.5,
                        marker='o', markersize=3, label=f'N = {N}')
        axes[1, 1].semilogy(x, np.abs(tau), color=color, lw=1.5,
                            marker='o', markersize=3, label=f'N = {N}')

    for ax in axes.flat:
        ax.axvline(-a_half, color='gray', ls=':', lw=0.8, alpha=0.6)
        ax.axvline(+a_half, color='gray', ls=':', lw=0.8, alpha=0.6)
        ax.axvline(0,       color='gray', ls=':', lw=0.6, alpha=0.4)
        ax.grid(True, alpha=0.3)
        ax.set_xlabel(r'$x_1$', fontsize=11)
        ax.legend(fontsize=9)

    axes[0, 0].set_ylabel(r'Re $\tau(x_1)$')
    axes[0, 0].set_title('Re τ')
    axes[0, 0].axhline(0, color='gray', lw=0.5)

    axes[0, 1].set_ylabel(r'Im $\tau(x_1)$')
    axes[0, 1].set_title('Im τ')
    axes[0, 1].axhline(0, color='gray', lw=0.5)

    axes[1, 0].set_ylabel(r'$|\tau(x_1)|$')
    axes[1, 0].set_title('|τ|  (линейная шкала)')

    axes[1, 1].set_ylabel(r'$|\tau(x_1)|$  (лог.)')
    axes[1, 1].set_title('|τ|  (логарифмическая шкала)')

    fig.suptitle(
        rf'BEM A: $\tau(x_1)$ — эталон, RK4 без асимптотик  '
        rf'($\kappa = {kappa}$, мягкий верх, $a = {a_half}$)',
        fontsize=13)
    plt.tight_layout()
    plt.savefig('bem_a_tau.png', dpi=200, bbox_inches='tight')
    print(f"\n✓ График сохранён в bem_a_tau.png")
    plt.show()


if __name__ == "__main__":
    main()
